    async function sendChat() {
        const input = document.getElementById('chat-input');
        const text = input.value.trim();
        if (!text) return;
        const box = document.getElementById('chat-container');
        box.innerHTML += `<div class="chat-msg user-msg">${text}</div>`;
        chatHistory.push({ role: "user", parts: [{ text }] });
        input.value = "";
        try {
            const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${modelId}:generateContent?key=${apiKey}`, {
                method: "POST", headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ contents: chatHistory, systemInstruction: { parts: [{ text: "你是四國導遊。熟知小豆島單車租借、道後溫泉收費、Happy House (HM5D8TQSBZ)、1/11 觀潮、1/14 下灘日落。嚴禁使用 Emoji。" }] } })
            });
            const data = await response.json();
            const aiText = data.candidates?.[0]?.content?.parts?.[0]?.text || "...";
            box.innerHTML += `<div class="chat-msg ai-msg">${aiText.replace(/\n/g, '<br>')}</div>`;
            chatHistory.push({ role: "model", parts: [{ text: aiText }] });
            box.scrollTop = box.scrollHeight;
        } catch (e) { box.innerHTML += `<div class="chat-msg ai-msg text-red-400">連線異常</div>`; }
    }
